1. **Page overflow after final figure placement** — content must end on page 8 even after floats settle; Task 13 Step 2 measures the page on which "REFERENCES" starts.

3. **Judge verdict inconsistent with its rationale** (seen in the March pilot: rationale said "correct" while `overall_correct=false`) — Task 1 Step 5 flags records whose rationale contains "should be true"/"all criteria are met" but `overall_correct` is false, and Task 1 uses two judges with agreement reported.

- [ ] **Step 4: Implement `scripts/eval/judge_summary.py`**

```python
#!/usr/bin/env python3
"""Summarize judge caches produced by scripts/judge_mekh_processes.py.

Usage:
  python scripts/eval/judge_summary.py --out-dir "$DATA" \
      --cache boron:gpt "$DATA/judge_boron_gpt.jsonl" \
      --cache boron:claude "$DATA/judge_boron_claude.jsonl" ...
Writes judge_summary.csv, judge_agreement.csv, judge_summary.md.
"""
from __future__ import annotations
import argparse, csv, json, random, re
from pathlib import Path

INCONSISTENT = re.compile(r"(should be true|all criteria are met|overall_correct should be true)", re.I)

def _rows(path: Path) -> dict[str, dict]:
    out = {}
    with open(path) as f:
        for line in f:
            if line.strip():
                r = json.loads(line); out[r["key"]] = r
    return out

def bootstrap_ci(bits: list[int], n_boot: int = 2000, seed: int = 0) -> tuple[float, float]:
    rng = random.Random(seed); n = len(bits)
    if n == 0: return (0.0, 0.0)
    means = sorted(sum(rng.choice(bits) for _ in range(n)) / n for _ in range(n_boot))
    return (means[int(0.025 * n_boot)], means[int(0.975 * n_boot) - 1])

def summarize_cache(path: Path) -> dict:
    rows = list(_rows(path).values())
    bits = [1 if r.get("overall_correct") else 0 for r in rows]
    lo, hi = bootstrap_ci(bits)
    rate = lambda k: (sum(1 for r in rows if r.get(k)) / len(rows)) if rows else 0.0
    return {"n": len(rows), "overall_correct": sum(bits),
            "precision": (sum(bits) / len(bits)) if bits else 0.0, "ci_lo": lo, "ci_hi": hi,
            "plausible_rate": rate("process_plausible"), "io_rate": rate("inputs_outputs_correct"),
            "hs_rate": rate("hs_codes_correct"),
            "inconsistent_count": sum(1 for r in rows if not r.get("overall_correct")
                                      and INCONSISTENT.search(r.get("rationale", "") or ""))}

def cohen_kappa(a: Path, b: Path) -> tuple[float, float]:
    ra, rb = _rows(a), _rows(b); keys = sorted(set(ra) & set(rb))
    if not keys: return (0.0, 0.0)
    xa = [bool(ra[k].get("overall_correct")) for k in keys]
    xb = [bool(rb[k].get("overall_correct")) for k in keys]
    n = len(keys); po = sum(1 for i in range(n) if xa[i] == xb[i]) / n
    pa, pb = sum(xa) / n, sum(xb) / n
    pe = pa * pb + (1 - pa) * (1 - pb)
    kappa = 0.0 if pe == 1 else (po - pe) / (1 - pe)
    return (po, kappa)

def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--cache", nargs=2, action="append", metavar=("LABEL:JUDGE", "PATH"), required=True)
    ap.add_argument("--out-dir", type=Path, required=True)
    a = ap.parse_args()
    caches = {tuple(lj.split(":", 1)): Path(p) for lj, p in a.cache}
    with open(a.out_dir / "judge_summary.csv", "w", newline="") as f:
        w = csv.writer(f); w.writerow(["label", "judge", "n", "overall_correct", "precision", "ci_lo", "ci_hi",
                                       "plausible_rate", "io_rate", "hs_rate", "inconsistent_count"])
        for (label, judge), p in sorted(caches.items()):
            s = summarize_cache(p)
            w.writerow([label, judge] + [s[k] for k in ["n", "overall_correct", "precision", "ci_lo", "ci_hi",
                                                          "plausible_rate", "io_rate", "hs_rate", "inconsistent_count"]])
    with open(a.out_dir / "judge_agreement.csv", "w", newline="") as f:
        w = csv.writer(f); w.writerow(["label", "n_both", "agreement", "cohen_kappa"])
        for label in sorted({l for l, _ in caches}):
            js = [caches[(l, j)] for (l, j) in sorted(caches) if l == label]
            if len(js) >= 2:
                po, k = cohen_kappa(js[0], js[1])
                w.writerow([label, len(set(_rows(js[0])) & set(_rows(js[1]))), f"{po:.3f}", f"{k:.3f}"])
    md = ["# Judge summary", "", "| label | judge | n | precision | 95% CI | plausible | I/O | HS | inconsistent |", "|---|---|--:|--:|--|--:|--:|--:|--:|"]
    for (label, judge), p in sorted(caches.items()):
        s = summarize_cache(p)
        md.append(f"| {label} | {judge} | {s['n']} | {s['precision']:.3f} | [{s['ci_lo']:.3f}, {s['ci_hi']:.3f}] | "
                  f"{s['plausible_rate']:.3f} | {s['io_rate']:.3f} | {s['hs_rate']:.3f} | {s['inconsistent_count']} |")
    (a.out_dir / "judge_summary.md").write_text("\n".join(md) + "\n")

if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""Evidence-support rate: does at least one cited reference mention the process's inputs and outputs?

Usage: python scripts/eval/evidence_support.py --out-dir "$DATA" --folder boron "$STATE_B" [--folder gallium "$STATE_GA" ...]
"""
from __future__ import annotations
import argparse, csv, hashlib, json, re
from pathlib import Path

def _norm(s: str) -> str:
    return re.sub(r"\s+", " ", re.sub(r"[^a-z0-9+\- ]", " ", s.lower())).strip()

def _ref_url(ref) -> str:
    return ref if isinstance(ref, str) else (ref.get("url") or "")

def material_terms(m, materials: dict) -> set[str]:
    """Names + aliases for a precursor/product entry; falls back to the inline name or bare string."""
    if isinstance(m, str):
        return {_norm(m)} - {""}
    terms = {_norm(m.get("name", ""))}
    rec = materials.get(str(m.get("hs_code", "")))
    if rec:
        terms.add(_norm(rec.get("name", "")))
        terms.update(_norm(a) for a in rec.get("aliases", []) or [])
    return {t for t in terms if len(t) >= 3}

def _mentions(text: str, terms: set[str]) -> bool:
    return any(t in text for t in terms)

def process_support(texts: list[str], pre_terms: list[set[str]], prod_terms: list[set[str]]) -> tuple[bool, bool]:
    """Returns (any_mention, fully_grounded). fully_grounded: one reference mentions >=1 precursor AND >=1 product."""
    any_m = full = False
    for raw in texts:
        t = _norm(raw or "")
        if not t: continue
        p_hit = any(_mentions(t, s) for s in pre_terms)
        q_hit = any(_mentions(t, s) for s in prod_terms)
        any_m |= (p_hit or q_hit); full |= (p_hit and q_hit)
    return any_m, full

def load_content(folder: Path, url: str) -> str | None:
    fn = folder / "reference_content" / (hashlib.md5(url.split("#", 1)[0].encode("utf-8")).hexdigest() + ".json")
    if not fn.exists(): return None
    try:
        return json.loads(fn.read_text()).get("content") or None
    except json.JSONDecodeError:
        return None

def evaluate_folder(folder: Path) -> tuple[dict, list[dict]]:
    state = json.loads((folder / "research_state.json").read_text())
    materials = state.get("materials", {}); procs = state.get("processes", {})
    rows, n_refs, refs_ok, unresolved = [], 0, 0, 0
    for pid, p in procs.items():
        pre = [material_terms(m, materials) for m in p.get("precursors", [])]
        prod = [material_terms(m, materials) for m in p.get("products", [])]
        unresolved += sum(1 for m in p.get("precursors", []) + p.get("products", [])
                          if isinstance(m, str) or str(m.get("hs_code", "")) not in materials)
        texts = []
        for r in p.get("references", []):
            n_refs += 1; c = load_content(folder, _ref_url(r))
            if c: refs_ok += 1; texts.append(c)
        any_m, full = process_support(texts, pre, prod)
        rows.append({"process_id": p.get("id", pid), "n_refs": len(p.get("references", [])),
                     "refs_with_content": len(texts), "any_mention": any_m, "fully_grounded": full})
    n = len(rows)
    summary = {"n_processes": n, "n_refs": n_refs, "refs_with_content": refs_ok,
               "frac_refs_with_content": refs_ok / n_refs if n_refs else 0.0,
               "proc_any_mention": sum(r["any_mention"] for r in rows),
               "proc_fully_grounded": sum(r["fully_grounded"] for r in rows),
               "frac_any": (sum(r["any_mention"] for r in rows) / n) if n else 0.0,
               "frac_full": (sum(r["fully_grounded"] for r in rows) / n) if n else 0.0,
               "unresolved_materials": unresolved}
    return summary, rows

def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--folder", nargs=2, action="append", metavar=("LABEL", "PATH"), required=True)
    ap.add_argument("--out-dir", type=Path, required=True)
    a = ap.parse_args()
    keys = ["n_processes", "n_refs", "refs_with_content", "frac_refs_with_content", "proc_any_mention",
            "proc_fully_grounded", "frac_any", "frac_full", "unresolved_materials"]
    with open(a.out_dir / "evidence_support.csv", "w", newline="") as f, \
         open(a.out_dir / "evidence_support_per_process.csv", "w", newline="") as g:
        w = csv.writer(f); w.writerow(["label"] + keys)
        wp = csv.writer(g); wp.writerow(["label", "process_id", "n_refs", "refs_with_content", "any_mention", "fully_grounded"])
        for label, path in a.folder:
            s, rows = evaluate_folder(Path(path)); w.writerow([label] + [s[k] for k in keys])
            for r in rows: wp.writerow([label] + [r[k] for k in ["process_id", "n_refs", "refs_with_content", "any_mention", "fully_grounded"]])
            print(label, s)

if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""Recall of author-curated canonical routes in each MEKH.

Usage: python scripts/eval/known_route_recall.py --routes "$DATA/known_routes.csv" --out-dir "$DATA" \
          --folder boron "$STATE_B" [--folder gallium "$STATE_GA" ...]
"""
from __future__ import annotations
import argparse, csv, json, re
from pathlib import Path

SIX = re.compile(r"^\d{6}$")

def load_routes(path: Path) -> tuple[list[dict], list[str]]:
    routes, rejected = [], []
    with open(path, newline="") as f:
        for r in csv.DictReader(f):
            ins = {c.strip() for c in r["input_hs"].split("|") if c.strip()}
            outs = {c.strip() for c in r["output_hs"].split("|") if c.strip()}
            if not ins or not outs or not all(SIX.match(c) for c in ins | outs):
                rejected.append(r["route_id"]); continue
            routes.append({"base_material": r["base_material"], "route_id": r["route_id"],
                           "description": r["description"], "inputs": ins, "outputs": outs})
    return routes, rejected

def process_hs_sets(folder: Path) -> list[tuple[set[str], set[str]]]:
    state = json.loads((folder / "research_state.json").read_text())
    out = []
    for p in state.get("processes", {}).values():
        hs = lambda ms: {str(m["hs_code"]) for m in ms if isinstance(m, dict) and m.get("hs_code")}
        out.append((hs(p.get("precursors", [])), hs(p.get("products", []))))
    return out

def recall_for(routes: list[dict], procs: list[tuple[set[str], set[str]]]) -> tuple[dict, dict]:
    detail = {}
    for r in routes:
        status = "miss"
        for pin, pout in procs:
            if r["inputs"] <= pin and r["outputs"] <= pout: status = "strict"; break
            if (r["inputs"] & pin) and (r["outputs"] & pout): status = "lenient"
        detail[r["route_id"]] = status
    n = len(routes); s_ = sum(1 for v in detail.values() if v == "strict")
    l_ = sum(1 for v in detail.values() if v in ("strict", "lenient"))
    return ({"n_routes": n, "recovered_lenient": l_, "recall_lenient": l_ / n if n else 0.0,
             "recovered_strict": s_, "recall_strict": s_ / n if n else 0.0}, detail)

def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--routes", type=Path, required=True)
    ap.add_argument("--folder", nargs=2, action="append", metavar=("LABEL", "PATH"), required=True)
    ap.add_argument("--out-dir", type=Path, required=True)
    a = ap.parse_args()
    routes, rejected = load_routes(a.routes)
    if rejected: print("REJECTED (non-6-digit or empty):", rejected)
    with open(a.out_dir / "known_route_recall.csv", "w", newline="") as f, \
         open(a.out_dir / "known_route_recall_detail.csv", "w", newline="") as g:
        w = csv.writer(f); w.writerow(["label", "n_routes", "recovered_lenient", "recall_lenient", "recovered_strict", "recall_strict"])
        wd = csv.writer(g); wd.writerow(["label", "route_id", "description", "status"])
        for label, path in a.folder:
            rs = [r for r in routes if r["base_material"] == label]
            s, detail = recall_for(rs, process_hs_sets(Path(path)))
            w.writerow([label] + [s[k] for k in ["n_routes", "recovered_lenient", "recall_lenient", "recovered_strict", "recall_strict"]])
            for r in rs: wd.writerow([label, r["route_id"], r["description"], detail[r["route_id"]]])
            print(label, s)

if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""RQ2: HS codes present in a MEKH but absent from the USGS Mineral Commodity Summaries listing.

Usage: python scripts/eval/topdown_coverage.py --usgs "$DATA/usgs_mcs_hs_codes.csv" --out-dir "$DATA" \
          --folder boron "$STATE_B" [--folder gallium "$STATE_GA" ...]
"""
from __future__ import annotations
import argparse, csv, json
from collections import Counter
from pathlib import Path

def mekh_codes(folder: Path) -> dict[str, tuple[str, int]]:
    state = json.loads((folder / "research_state.json").read_text())
    part = Counter()  # number of distinct processes each code participates in
    for p in state.get("processes", {}).values():
        codes = {str(m["hs_code"]) for m in p.get("precursors", []) + p.get("products", [])
                 if isinstance(m, dict) and m.get("hs_code")}
        for c in codes: part[c] += 1
    return {code: (rec.get("name", ""), part.get(code, 0)) for code, rec in state.get("materials", {}).items()}

def coverage(codes: dict[str, tuple[str, int]], usgs: set[str]) -> tuple[dict, list[tuple[str, str, int]]]:
    only = sorted(((c, n, k) for c, (n, k) in codes.items() if c not in usgs), key=lambda t: -t[2])
    s = {"mekh_codes": len(codes), "usgs_codes": len(usgs), "overlap": len(set(codes) & usgs),
         "mekh_only": len(only), "frac_mekh_only": (len(only) / len(codes)) if codes else 0.0}
    return s, only

def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--usgs", type=Path, required=True)
    ap.add_argument("--folder", nargs=2, action="append", metavar=("LABEL", "PATH"), required=True)
    ap.add_argument("--out-dir", type=Path, required=True)
    a = ap.parse_args()
    usgs: dict[str, set[str]] = {}
    with open(a.usgs, newline="") as f:
        for r in csv.DictReader(f): usgs.setdefault(r["commodity"], set()).add(r["hs_code"].strip())
    with open(a.out_dir / "topdown_coverage.csv", "w", newline="") as f, \
         open(a.out_dir / "topdown_coverage_examples.csv", "w", newline="") as g:
        w = csv.writer(f); w.writerow(["label", "mekh_codes", "usgs_codes", "overlap", "mekh_only", "frac_mekh_only"])
        we = csv.writer(g); we.writerow(["label", "hs_code", "name", "n_processes"])
```

```python
#!/usr/bin/env python3
"""Turn eval CSVs into LaTeX macros (eval_numbers.tex) and tables (eval_tables.tex).

Usage: python scripts/eval/make_latex.py --data-dir "$DATA"
"""
from __future__ import annotations
import argparse, csv
from pathlib import Path

def cap(label: str) -> str: return label[:1].upper() + label[1:]
def pct(x: float) -> str: return f"{round(float(x) * 100)}\\%"
def macro(name: str, label: str, value) -> str: return f"\\newcommand{{\\{name}{cap(label)}}}{{{value}}}"
def _read(p: Path) -> list[dict]:
    return list(csv.DictReader(open(p, newline=""))) if p.exists() else []

def build_numbers(d: Path) -> str:
    L = []
    js = _read(d / "judge_summary.csv")
    for r in js:
        j = cap(r["judge"]); l = r["label"]
        L += [macro("JudgeN", l, r["n"]), macro("JudgePrecision", l + j, pct(r["precision"])),
              macro("JudgeCILo", l + j, pct(r["ci_lo"])), macro("JudgeCIHi", l + j, pct(r["ci_hi"])),
              macro("JudgePlausible", l + j, pct(r["plausible_rate"])), macro("JudgeIO", l + j, pct(r["io_rate"])),
              macro("JudgeHS", l + j, pct(r["hs_rate"]))]
    for judge in sorted({r["judge"] for r in js}):
        rows = [r for r in js if r["judge"] == judge]; n = sum(int(r["n"]) for r in rows)
        ok = sum(int(r["overall_correct"]) for r in rows)
        L += [macro("JudgePrecision", "all" + cap(judge), pct(ok / n if n else 0.0)), macro("JudgeN", "all", n)]
    for r in _read(d / "judge_agreement.csv"):
        L += [macro("JudgeKappa", r["label"], f"{float(r['cohen_kappa']):.2f}"), macro("JudgeAgreement", r["label"], pct(r["agreement"]))]
    for r in _read(d / "evidence_support.csv"):
        L += [macro("EvidenceFull", r["label"], pct(r["frac_full"])), macro("EvidenceAny", r["label"], pct(r["frac_any"])),
              macro("EvidenceRefsFetched", r["label"], pct(r["frac_refs_with_content"])), macro("EvidenceNRefs", r["label"], r["n_refs"])]
    for r in _read(d / "known_route_recall.csv"):
        L += [macro("RoutesN", r["label"], r["n_routes"]), macro("RecallLenient", r["label"], pct(r["recall_lenient"])),
              macro("RecallStrict", r["label"], pct(r["recall_strict"]))]
    for r in _read(d / "topdown_coverage.csv"):
        L += [macro("CoverageMekhOnly", r["label"], r["mekh_only"]), macro("CoverageFrac", r["label"], pct(r["frac_mekh_only"])),
              macro("CoverageUsgs", r["label"], r["usgs_codes"]), macro("CoverageMekh", r["label"], r["mekh_codes"])]
    # \newcommand errors on redefinition, and JudgeN<Label> / JudgeNAll are emitted once per judge: keep the first of each name.
    seen: dict[str, str] = {}
    for line in L:
        name = line.split("}{", 1)[0]
        seen.setdefault(name, line)
    return "% generated by scripts/eval/make_latex.py -- do not edit\n" + "\n".join(seen.values()) + "\n"

def build_tables(d: Path) -> str:
    T = []
    js = _read(d / "judge_summary.csv"); ag = {r["label"]: r for r in _read(d / "judge_agreement.csv")}
    if js:
        T += [r"\begin{table}[t]\centering\small", r"\caption{RQ1 precision proxy: fraction of hyperedges judged fully correct (process, inputs/outputs, HS codes) by two independent LLM judges, with bootstrap 95\% CIs and inter-judge agreement ($\kappa$).}",
              r"\label{tab:judge}", r"\begin{tabular}{llrrrr}\toprule", r"MEKH & Judge & $n$ & Precision & 95\% CI & $\kappa$ \\\midrule"]
        for r in js:
            T.append(f"{cap(r['label'])} & {r['judge']} & {r['n']} & {pct(r['precision'])} & [{pct(r['ci_lo'])}, {pct(r['ci_hi'])}] & {float(ag[r['label']]['cohen_kappa']):.2f} \\\\" if r['label'] in ag
                     else f"{cap(r['label'])} & {r['judge']} & {r['n']} & {pct(r['precision'])} & [{pct(r['ci_lo'])}, {pct(r['ci_hi'])}] & -- \\\\")
        T += [r"\bottomrule\end{tabular}\end{table}", ""]
    ev = _read(d / "evidence_support.csv")
    if ev:
        T += [r"\begin{table}[t]\centering\small", r"\caption{Evidence grounding: share of hyperedges whose cited references mention at least one input and one output (fully grounded), any input or output, and share of references with retrievable content.}",
              r"\label{tab:evidence}", r"\begin{tabular}{lrrrr}\toprule", r"MEKH & Refs & Retrievable & Any mention & Fully grounded \\\midrule"]
        T += [f"{cap(r['label'])} & {r['n_refs']} & {pct(r['frac_refs_with_content'])} & {pct(r['frac_any'])} & {pct(r['frac_full'])} \\\\" for r in ev]
        T += [r"\bottomrule\end{tabular}\end{table}", ""]
    rc = _read(d / "known_route_recall.csv")
    if rc:
        T += [r"\begin{table}[t]\centering\small", r"\caption{Recall of author-curated canonical routes (lenient: shared input and output code; strict: route codes $\subseteq$ hyperedge codes).}",
              r"\label{tab:recall}", r"\begin{tabular}{lrrr}\toprule", r"MEKH & Routes & Lenient & Strict \\\midrule"]
        T += [f"{cap(r['label'])} & {r['n_routes']} & {pct(r['recall_lenient'])} & {pct(r['recall_strict'])} \\\\" for r in rc]
        T += [r"\bottomrule\end{tabular}\end{table}", ""]
    cv = _read(d / "topdown_coverage.csv")

              r"\label{tab:coverage}", r"\begin{tabular}{lrrrr}\toprule", r"MEKH & MEKH codes & USGS codes & Overlap & MEKH-only \\\midrule"]

        T += [r"\bottomrule\end{tabular}\end{table}", ""]
```

```latex
\section{Introduction}
\label{sec:introduction}

Advanced technologies depend on long chains of industrial transformations that turn ores, brines, and mineral concentrates into intermediate chemicals and finally into components. Where those chains concentrate in a single supplier or country, they become points of leverage and of failure; recent semiconductor, rare-earth, and battery-material shortages all originated at intermediate processing stages rather than at the mine or the finished product~\cite{hasan2022demand,zhang2024reshaping,hu2025technological}. Analysts who must anticipate such disruptions predominantly work \emph{top-down}: they start from a product and trace backward through supplier declarations and trade records. Baldwin et al.~\cite{baldwin2023hidden} show that this systematically underestimates exposure, because the inputs used to make inputs are hidden behind proprietary processes and opaque supplier networks~\cite{colon2022systemic}.

\paragraph{Application context.}
This work was carried out with, and is used by, supply-chain analysts at a university national-security policy research institute working under contract to a U.S. government sponsor. Their recurring question is concrete: for a given critical mineral, \emph{which intermediate materials and processes lie between the mine and the technology, which of them are concentrated in a few exporters, and what fails downstream if one is cut off?} Answering it requires a map of material transformations that is (i) typed by the Harmonized System (HS) codes in which trade is actually recorded, so that it can be joined to trade data, and (ii) traceable to documentary evidence, so that an analyst can audit any link before acting on it. No such map exists for most critical minerals, and building one by hand does not scale. The same HS-typed classification that the map is built on also feeds the analysts' other tools: it supplies the commodity-to-code buckets that join trade-flow data to their dependency-network explorer (Section~\ref{sec:application}).

\paragraph{Approach.}
We built MP-DNF, an agentic pipeline that constructs such a map \emph{bottom-up}. Starting from a seed mineral, six role-specialized LLM agents---classification, material finding, process discovery, process merging, reference review, and source scoring---propose materials and industrial processes, retrieve supporting documents through Model Context Protocol (MCP) tools, and score the evidence. The output is a \emph{Material Evolution Knowledge Hypergraph} (MEKH): a directed labeled multi-hypergraph whose vertices are HS-6-coded materials and whose hyperedges are processes mapping input sets to output sets (Section~\ref{sec:mekh}). The agents propose; deterministic gates dispose. A process enters the MEKH only if its materials resolve to valid HS-6 codes and it is supported by at least two references from distinct domains whose relevance and source quality clear a threshold (Section~\ref{sec:pipeline}). Every hyperedge therefore carries the evidence an analyst needs to audit it.

\paragraph{Research questions.}
An earlier presentation of this system reported network statistics on the constructed hypergraphs but did not establish whether the hypergraphs were correct or whether bottom-up discovery adds anything over a top-down view. We address that directly:
\begin{itemize}[leftmargin=*,nosep]
  \item \textbf{RQ1 (accuracy and grounding).} Are the discovered transformations factually correct---process, inputs and outputs, HS codes---and are they actually supported by the references cited?
  \item \textbf{RQ2 (coverage beyond top-down).} Does bottom-up discovery surface intermediate materials that the official top-down commodity view omits?
  \item \textbf{RQ3 (robustness).} Are the criticality rankings derived from a MEKH stable under the stochasticity of LLM-driven generation?
\end{itemize}

\paragraph{Contributions.}
(1)~A deployed agentic pipeline, MP-DNF, in which LLM agents with MCP tools construct evidence-grounded, HS-typed knowledge hypergraphs of material transformations, with explicit symbolic admission gates (Section~\ref{sec:pipeline}).
(2)~An evaluation that requires no expert annotation and answers RQ1--RQ3 for MEKHs of boron, gallium, germanium, and cobalt: a two-judge LLM precision proxy with bootstrap confidence intervals and inter-judge agreement, an evidence-support audit of cited references, recall against curated canonical routes, and a comparison with the USGS commodity listing (Section~\ref{sec:evaluation}).
(3)~A criticality measure on hypergraph reachability and a sensitivity method based on rank aggregation, and a disruption analysis coupling the boron MEKH to UN Comtrade export data that localizes cascading risk to a few concentrated intermediate codes (Section~\ref{sec:application}).
(4)~Lessons from deployment with analysts, including failure modes of LLM-generated supply-chain knowledge and the limits of LLM-judge validation (Section~\ref{sec:lessons}).
```

```latex
\section{Related Work}
\label{sec:related}

\textbf{LLM-based knowledge graph construction.}
Large language models are now widely used to extract entities and relations into knowledge graphs, from biomedical records~\cite{arsenyan2024biomedical} to general-purpose pipelines surveyed by Pan et al.~\cite{pan2024unifying} and, for graph analytics more broadly, by Shang and Huang~\cite{shang2025survey}. Yang et al.~\cite{yang2024inductive} show that LLMs can induce rules from facts, and Chu et al.~\cite{chu2024llmhypergraph} use LLM guidance to build hypergraphs for recommendation. These systems extract from a given corpus; MP-DNF instead \emph{discovers} the corpus as it goes, using agents with retrieval tools to find and score evidence for each candidate relation. AlMahri et al.~\cite{almahri2024enhancing} apply LLM extraction to supply-chain visibility from existing documentation, and Liu et al.~\cite{liu2025designing} integrate existing supply-chain data through knowledge structures; neither constructs transformation networks from raw materials, and neither types nodes by trade codes.

\textbf{Tool-using and multi-agent LLM systems.}
Our agents follow the reason-and-act pattern~\cite{yao2023react} with tools exposed through the Model Context Protocol~\cite{mcp2024}. Multi-agent systems have long been applied to supply-chain coordination and operations~\cite{jaimez2021towards,farazi2025enhancing,santos2024implementing,vanderheijden2024multiagent}; those agents make operational decisions, whereas ours perform knowledge discovery and validation. The design of MP-DNF is a hybrid classic--generative one: generative agents propose, and deterministic symbolic gates (code validity, evidence quantity and diversity, score thresholds) decide admission.

\textbf{Hypergraphs and supply-chain networks.}
Industrial processes take several inputs to several outputs, which binary graphs represent poorly~\cite{wolf2016advantages,fatemi2019knowledge,luo2025hypergraphrag}. \section{Material Evolution Knowledge Hypergraphs}
```
```latex
\label{sec:mekh}

A \emph{Material Evolution Knowledge Hypergraph} (MEKH) is a directed labeled multi-hypergraph $\Hcal(M,E)$~\cite{ouvrardHypergraphsIntroductionReview2020}. Each vertex $m\in M$ is a material identified by a six-digit Harmonized System code~\cite{hscode}; HS-6 is the finest level at which codes are internationally comparable and at which UN Comtrade reports trade, so a MEKH can be joined directly to trade flows (Section~\ref{sec:application}). Each hyperedge $e=(e_s,e_t)\in E$ is an industrial or chemical process $\mathbf{P}_e$ transforming an input set $e_s\subset M$ into an output set $e_t\subset M$; $\ell(e)=\mathbf{P}_e$ labels it, and several processes with the same $(e_s,e_t)$ are distinct parallel hyperedges. A hyperedge carries a narrative description, a scale score, and a list of references, each with a source-quality score, a content-relevance score, and the URL from which an analyst can audit it.

Because vertices are HS-6 codes, purification steps whose input and output share a code (e.g., crude and electronic-grade germanium metal, both HS~811292) appear as self-loops; we keep them, as they are real chokepoints, but they are invisible to code-set analyses.

Algorithm~\ref{alg:tdn-pipeline} gives the outer loop. Starting from seed materials, each iteration calls \textsc{ProcessGeneration}, which queries auxiliary resources $A$ (HS-code search, web search, encyclopedic sources, and the reference cache) for processes that produce or consume known materials, admits those passing the gates of Section~\ref{sec:gates}, and adds their inputs and outputs as new vertices.
```

```latex
\subsection{Admission gates}
\label{sec:gates}
A candidate process becomes a hyperedge only if: (G1) every precursor and product resolves to a valid HS-6 code; (G2) it cites at least $k_{\min}=2$ references from at least $d_{\min}=2$ distinct web domains; (G3) its process score is defined and $\ge\tau$; and (G4) its scale score meets the configured minimum. Processes failing G2--G4 after the maximum number of review rounds are removed by the \emph{purge} task, the only task with no LLM. Gates are the reason the pipeline's errors are auditable: an analyst who doubts a hyperedge can open its references and see why it was admitted.

\subsection{Iteration and modes}
Process discovery runs in \emph{create} mode for materials with no processes and in \emph{edit} mode otherwise; in edit mode the agent sees the existing processes and their reference scores and may add references, revise inputs/outputs, or remove a process it now judges unsupported. Upstream/downstream expansion sends every unresolved precursor or product to the material finder, and the next iteration discovers processes for the new materials. Iteration stops after $n$ rounds or when no new materials are found.
```

```latex
\section{Deployment, Lessons Learned, and Limitations}
\label{sec:lessons}

\paragraph{Deployment.}
MP-DNF runs as a command-line pipeline with resumable state, so an analyst can seed a mineral, let discovery run overnight, inspect the MEKH and its references, add or remove materials, and resume. Reference content is cached and indexed for semantic search, so later iterations and later minerals reuse evidence already fetched. The analysts' requests shaped three design decisions that a research prototype would not have made: HS-6 typing of every vertex (for the trade-data join), mandatory per-hyperedge references with scores (for auditability), and a purge step that removes rather than flags unsupported processes (analysts preferred a smaller trusted graph to a larger annotated one).

\paragraph{What goes wrong.}
Three failure modes dominate. (1)~\emph{Code drift}: the LLM names the right material but the HS search returns a neighbouring subheading; this is the main source of RQ1 errors and is best fixed by constraining code assignment to the subheadings of a heading already in the graph. (2)~\emph{Reagent omission}: reducing agents and acids are described in the narrative but not listed as precursors, lowering strict recall; a schema field for reagents would recover this. (3)~\emph{Generic-process drift}: an agent asked about a boron intermediate may return a general process (e.g., sulfuric-acid manufacture) that is true but off-topic; the distinct-domain evidence gate does not stop it, and a relevance-to-seed score would.

\paragraph{Limitations.}
The LLM judges are a proxy for expert review; they agree with each other ($\kappa$ in Table~\ref{tab:judge}) but may share blind spots, and we report their verdicts as given even where a rationale and verdict disagree. Evidence grounding is measured by string matching and under-counts paraphrase. Canonical-route recall depends on an author-curated set that is small and biased toward well-documented routes. HS-6 aggregation merges materials of different purity, and five iterations bound depth. All four MEKHs are in one domain; generality of the architecture is a design claim, not an empirical one.

\paragraph{Ethical and responsible-use considerations.}
The pipeline reads only public sources and produces analyses of public trade data; it does not model firms or individuals. Its outputs inform policy analysis and are audited by analysts before use, which the reference requirement is designed to support. Generative-AI use in this work is confined to the agents that are the object of study and to editorial polishing of the text; all analyses and claims are the authors' responsibility.

\paragraph{Availability.}
Code is available in an anonymized repository (\url{https://anonymous.4open.science/r/lia-B773}); agent prompts, evaluation scripts, judge outputs, the curated route and USGS-code files, and the MEKH state files are in the supplementary material and will be archived on Zenodo on publication.
```
